Console will automatically create hosted-aws/pull-secret.json from this content
Console will automatically create hosted-aws/sts-token.json from this content
./hcp create cluster aws \
  --name virt-hcp-maestro-1 \
  --target-cluster mce-1 \
  --maestro-server https://maestro-server.example.com \
  --maestro-consumer mce-1 \
  --pull-secret hosted-aws/pull-secret.json \
  --sts-creds hosted-aws/sts-token.json \
  --node-pool-replicas 1 \
  --base-domain dev08.red-chesterfield.com \
  --role-arn arn:aws:iam::273233148990:role/yikim-hosted \
  --release-image quay.io/openshift-release-dev/ocp-release:4.19.0-multi \
  --control-plane-availability-policy SingleReplica \
  --external-dns-domain dev08.red-chesterfield.com \
  --etcd-storage-class managed-csi
